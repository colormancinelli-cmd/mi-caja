rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Solamente el usuario autenticado puede consultar sus datos.
    match /users/{userId} {
      allow read, write: if false;

      match /entries/{entryId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create, update: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.keys().hasOnly([
            'id','type','amount','description','date','category','account','toAccount','createdAt'
          ])
          && request.resource.data.id == entryId
          && request.resource.data.id is string
          && request.resource.data.type in ['expense','income','transfer']
          && request.resource.data.amount is int
          && request.resource.data.amount > 0
          && request.resource.data.amount < 10000000000000
          && request.resource.data.description is string
          && request.resource.data.description.size() <= 100
          && request.resource.data.date is string
          && request.resource.data.date.size() == 10
          && request.resource.data.category is string
          && request.resource.data.category.size() <= 100
          && request.resource.data.account in ['cash','bank','wallet']
          && request.resource.data.toAccount is string
          && request.resource.data.createdAt is string;
        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      match /meta/opening {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create, update: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.keys().hasOnly(['cash','bank','wallet'])
          && request.resource.data.keys().hasAll(['cash','bank','wallet'])
          && request.resource.data.cash is int
          && request.resource.data.bank is int
          && request.resource.data.wallet is int;
        allow delete: if false;
      }
    }
  }
}
